```typescript
// lib/algorithm/sovereignFeed.ts

interface FeedParameters {
  mood: 'contemplative' | 'energetic' | 'gentle' | 'chaotic' | 'focused';
  depth: 'surface' | 'deep' | 'comprehensive';
  socialRadius: 'self' | 'circle' | 'community' | 'global';
  noveltySeeking: 0-100; // 0 = familiar, 100 = novel
  affectiveState: 'seeking_comfort' | 'seeking_challenge' | 'neutral';
}

interface ContentScoring {
  postId: string;
  baseScore: number; // Computed from user params
  explanation: string; // Human-readable why
  transparencyLog: {
    contentTags: string[];
    userMoodMatch: number;
    socialProximity: number;
    freshness: number;
    randomness: number; // % of dice roll
  };
}

export function generateSovereignFeed(
  userParams: FeedParameters,
  contentPool: Post[],
  userGraph: SocialGraph
): ContentScoring[] {
  
  return contentPool.map(post => {
    // 1. Mood Matching (Transparent weights)
    const moodMatch = calculateMoodAlignment(post.tags, userParams.mood);
    
    // 2. Social Proximity (No shadow metrics)
    const proximity = getRelationshipDistance(userGraph, post.authorId);
    
    // 3. Temporal Decay (Chronological boost, not algorithmic burying)
    const freshness = hoursSince(post.createdAt) * -0.1; // Linear decay, not exponential
    
    // 4. Controlled Randomness (Serendipity injection)
    const chaosDie = Math.random() * userParams.noveltySeeking;
    
    // 5. The Explanation (Why am I seeing this?)
    const explanation = buildExplanation({
      moodMatch,
      proximity,
      freshness,
      chaosDie,
      userParams
    });
    
    return {
      postId: post.id,
      baseScore: (moodMatch * 0.4) + (proximity * 0.3) + (freshness * 0.2) + (chaosDie * 0.1),
      explanation,
      transparencyLog: {
        contentTags: post.tags,
        userMoodMatch: moodMatch,
        socialProximity: proximity,
        freshness,
        randomness: chaosDie
      }
    };
  }).sort((a, b) => b.baseScore - a.baseScore);
}

function buildExplanation(log): string {
  const reasons = [];
  if (log.moodMatch > 0.8) reasons.push("matches your requested mood");
  if (log.socialProximity < 2) reasons.push("from someone in your circle");
  if (log.freshness > 0.9) reasons.push("recently published");
  if (log.randomness > 50) reasons.push("added via serendipity dice roll");
  
  return `Showing this because it ${reasons.join(", ")}.`;
}
```
