-- User timeline events
CREATE TABLE user_timeline (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES profiles(id) ON DELETE CASCADE,
  event_type timeline_event_type NOT NULL,
  event_date TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  title TEXT NOT NULL,
  description TEXT,
  metadata JSONB DEFAULT '{}'::jsonb,
  significance_score INTEGER DEFAULT 0, -- 0-100, calculated importance
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_user_timeline_user_id ON user_timeline(user_id);
CREATE INDEX idx_user_timeline_event_date ON user_timeline(event_date);
CREATE INDEX idx_user_timeline_event_type ON user_timeline(event_type);

-- Timeline milestones (significant events)
CREATE TABLE timeline_milestones (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES profiles(id) ON DELETE CASCADE,
  milestone_date TIMESTAMPTZ NOT NULL,
  title TEXT NOT NULL,
  description TEXT,
  icon_name TEXT,
  color_hex TEXT,
  significance_score INTEGER DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()

  confidence pattern_confidence NOT NULL DEFAULT 'low',
  pattern_data JSONB NOT NULL, -- The pattern data
  insight TEXT, -- Human-readable insight
  occurred_at TIMESTAMPTZ NOT NULL,
  expires_at TIMESTAMPTZ, -- When this pattern observation expires
  created_at TIMESTAMPTZ DEFAULT NOW()

CREATE INDEX idx_recognized_patterns_pattern_type ON recognized_patterns(pattern_type);
CREATE INDEX idx_recognized_patterns_occurred_at ON recognized_patterns(occurred_at);

  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  pattern_data JSONB,

  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- ============================================================================
-- TIMELINE EVENT AUTO-CREATION
-- ============================================================================

-- Auto-create timeline events on quest completion
CREATE OR REPLACE FUNCTION auto_create_quest_timeline_event()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.status = 'completed' AND OLD.status != 'completed' THEN
    INSERT INTO user_timeline (user_id, event_type, title, description, metadata)
    SELECT 
      NEW.user_id,
      'quest_complete',
      q.title,
      q.description,
      jsonb_build_object('quest_id', NEW.quest_id, 'reward', q.sovereignty_reward)
    FROM quests q
    WHERE q.id = NEW.quest_id;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_quest_timeline_event
  AFTER UPDATE ON user_quests
  FOR EACH ROW
  EXECUTE FUNCTION auto_create_quest_timeline_event();

-- Auto-create timeline events on badge earning
CREATE OR REPLACE FUNCTION auto_create_badge_timeline_event()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO user_timeline (user_id, event_type, title, description, metadata)
  SELECT
