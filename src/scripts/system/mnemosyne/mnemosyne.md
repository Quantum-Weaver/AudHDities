| **Video Editor** | Timeline, cuts, transitions, effects | Video Core |
| **3D Forge** | Modeling, texturing, rendering | 3D Core |
| **Layout Weaver** | Page design, typography, grids | Image Core |

CREATE TABLE mnemosyne_modules (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  module_name TEXT NOT NULL UNIQUE,
  module_type TEXT NOT NULL, -- 'image', 'audio', 'video', '3d', 'text', 'code', 'layout', 'animation'
  version TEXT NOT NULL,
  capabilities TEXT[] NOT NULL, -- ['layers', 'filters', 'masks', ...]
  dependencies TEXT[], -- Other modules this requires
  wasm_url TEXT, -- URL to WebAssembly module
  js_url TEXT, -- URL to JavaScript wrapper
  css_url TEXT, -- URL to styles
  is_core BOOLEAN DEFAULT FALSE, -- Core modules load always
  is_active BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMPTZ DEFAULT NOW(),

);
